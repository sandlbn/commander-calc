% ---------------------------------------------------------------------------
% cclogo.tex -- the Commander Calc logo, redrawn in TikZ
%
% A coordinate-for-coordinate transcription of docs/commander-calc-logo.svg.
% The SVG viewBox is 0 0 1600 520; the picture below is drawn in those same
% units at 0.25pt each, then \resizebox scales the finished artwork -- which
% scales the rules and the type together and keeps the proportions exact.
%
% One deliberate difference from the SVG. There, CALC is a black panel with
% white stripes clipped to the glyph outlines. Clipping to a glyph is awkward
% in TikZ and unnecessary on a white page: drawing the word in ink and then
% laying white stripes across it produces the identical image.
% ---------------------------------------------------------------------------

% Letter-spaced small caps-ish setting, in thousandths of an em, matching the
% SVG's letter-spacing attribute divided by its font-size.
\providecommand{\ccls}[2]{\textls[#1]{#2}}

\newcommand{\cclogo}[1][3.2in]{%
\resizebox{#1}{!}{%
\begin{tikzpicture}[x=0.25pt,y=-0.25pt,line cap=butt,line join=miter]

  % ---- bounding box, tight to the ink, so \cclogo{w} really is w wide -----
  \path (83,63) rectangle (1499,467);

  % ======================= the spreadsheet symbol ========================
  % SVG group transform: translate(90,70). Coordinates below are absolute.

  % outer case
  \draw[ccink,line width=3.5pt,rounded corners=2pt] (90,70) rectangle (480,450);
  % inner frame
  \draw[ccink,line width=1.75pt] (116,97) rectangle (454,423);
  % column-header band
  \fill[ccblue] (116,97) rectangle (454,159);
  % row-header column
  \fill[ccink]  (116,97) rectangle (188,423);

  % grid rules
  \draw[ccink,line width=1.75pt] (188,97) -- (188,423);
  \draw[ccink,line width=1.75pt] (277,97) -- (277,423);
  \draw[ccink,line width=1.75pt] (366,97) -- (366,423);
  \draw[ccink,line width=1.75pt] (116,159) -- (454,159);
  \draw[ccink,line width=1.75pt] (116,225) -- (454,225);
  \draw[ccink,line width=1.75pt] (116,291) -- (454,291);
  \draw[ccink,line width=1.75pt] (116,357) -- (454,357);

  % the white ticks that read as column letters and row numbers
  \fill[white] (215,123) rectangle ++(34,10);
  \fill[white] (304,123) rectangle ++(34,10);
  \fill[white] (393,123) rectangle ++(34,10);
  \fill[white] (140,182) rectangle ++(24,9);
  \fill[white] (140,248) rectangle ++(24,9);
  \fill[white] (140,314) rectangle ++(24,9);
  \fill[white] (140,380) rectangle ++(24,9);

  % the cell cursor
  \fill[ccblue] (277,225) rectangle ++(89,66);
  \draw[ccink,line width=1.75pt] (277,225) rectangle ++(89,66);

  % ============================ the wordmark =============================

  \node[anchor=base west,inner sep=0pt,text=ccink] at (600,150)
    {\sffamily\bfseries\fontsize{19.5pt}{19.5pt}\selectfont\ccls{215}{COMMANDER}};

  \draw[ccink,line width=2pt] (600,185) -- (1495,185);

  % CALC, then the five stripes across it. The SVG is set in DejaVu Sans at
  % weight 800, which is appreciably wider and heavier than Nimbus Sans Bold;
  % stretching the word horizontally recovers both the width and the colour.
  \node[anchor=base west,inner sep=0pt,text=ccink] at (600,370)
    {\sffamily\bfseries\fontsize{62.5pt}{62.5pt}\selectfont
     \scalebox{1.28}[1]{\ccls{16}{CALC}}};
  \fill[white] (585,222) rectangle ++(930,12);
  \fill[white] (585,252) rectangle ++(930,12);
  \fill[white] (585,282) rectangle ++(930,12);
  \fill[white] (585,312) rectangle ++(930,12);
  \fill[white] (585,342) rectangle ++(930,12);

  \draw[ccink,line width=1.25pt] (600,414) -- (1495,414);

  \node[anchor=base west,inner sep=0pt,text=ccink] at (604,462)
    {\sffamily\bfseries\fontsize{8.5pt}{8.5pt}\selectfont
     \ccls{272}{SPREADSHEET FOR COMMANDER X16}};

\end{tikzpicture}}%
}
